#!/usr/bin/env python
# -*- coding: utf-8 -*-
import sys
from PySide6.QtWidgets import QApplication

from main_window import MainWindow
from libs import __version__

__appname__ = 'LabelCraft'


def get_main_app(argv=None):
    if argv is None:
        argv = sys.argv

    app = QApplication(argv)
    app.setApplicationName(__appname__)
    app.setWindowIcon(new_icon("app"))   # 假设你在 libs 中有 new_icon 函数

    # 保留你原来的 argparse 逻辑（未做修改）
    import argparse
    from os.path import dirname, join

    argparser = argparse.ArgumentParser(prog='labelcraft')
    argparser.add_argument("image_dir", nargs="?", help="Directory containing images")
    argparser.add_argument("--classes", "-c", dest="class_file",
                           default=join(dirname(__file__), "data", "predefined_classes.txt"))
    argparser.add_argument("--save-dir", "-s", dest="save_dir", nargs="?")
    argparser.add_argument("--version", "-v", action="version", version=f"{__appname__} {__version__}")

    args = argparser.parse_args(argv[1:])

    win = MainWindow(
        default_filename=args.image_dir,
        default_prefdef_class_file=args.class_file,
        default_save_dir=args.save_dir
    )

    win.setMinimumSize(800, 600)
    win.show()
    return app, win


if __name__ == '__main__':
    app, window = get_main_app()
    sys.exit(app.exec())

class WindowMixin:
    def menu(self, title, actions=None):
        menu = self.menuBar().addMenu(title)
        if actions:
            add_actions(menu, actions)
        return menu

    def toolbar(self, title, actions=None):
        toolbar = ToolBar(title)   # 假设你有 ToolBar 类
        toolbar.setObjectName(f'{title}ToolBar')
        toolbar.setToolButtonStyle(Qt.ToolButtonTextBesideIcon)
        toolbar.setMovable(False)
        if actions:
            add_actions(toolbar, actions)
        self.addToolBar(Qt.LeftToolBarArea, toolbar)
        return toolbar


class MainWindow(QMainWindow, WindowMixin):
    FIT_WINDOW, FIT_WIDTH, MANUAL_ZOOM = range(3)

    def __init__(self, default_filename=None, default_prefdef_class_file=None, default_save_dir=None):
        super().__init__()
        self.setWindowTitle(__appname__)

        # 基础设置
        self.settings = Settings()
        self.settings.load()
        self.os_name = platform.system()

        self.cur_img_idx = 0
        self.img_count = 0

        self.dirty = False

        self.file_path = None
        self.image = QImage()

        # 业务逻辑处理器
        self.handlers = MainWindowHandlers(self)

        # 项目管理器
        self.project_manager = ProjectManager(self)

        # 创建所有 Actions
        self.actions = create_all_actions(self)

        # 创建所有 UI 组件（右面板、工具栏、菜单、canvas 等）
        create_ui_components(self)

        # 连接信号
        self._connect_signals()

        # 初始化状态
        self._init_state(default_filename, default_prefdef_class_file, default_save_dir)

    def _connect_signals(self):
        """集中连接所有信号到 handlers"""
        self.canvas.newShape.connect(self.handlers.new_shape)
        self.canvas.shapeMoved.connect(self.handlers.set_dirty)
        self.canvas.selectionChanged.connect(self.handlers.shape_selection_changed)
        self.canvas.shapeDoubleClicked.connect(self.handlers.edit_label)
        self.canvas.drawingPolygon.connect(self.handlers.toggle_drawing_sensitive)

        self.label_list.itemActivated.connect(self.handlers.label_selection_changed)
        self.label_list.itemSelectionChanged.connect(self.handlers.label_selection_changed)
        self.label_list.itemDoubleClicked.connect(self.handlers.edit_label)
        self.label_list.itemChanged.connect(self.handlers.label_item_changed)

        self.file_list_widget.itemDoubleClicked.connect(self.handlers.file_item_double_clicked)
        self.pending_list_widget.itemDoubleClicked.connect(self.handlers.pending_item_double_clicked)

        self.zoom_widget.valueChanged.connect(self.handlers.paint_canvas)
        self.light_widget.valueChanged.connect(self.handlers.paint_canvas)

        # 项目相关信号可在这里继续添加

    def _init_state(self, default_filename, default_prefdef_class_file, default_save_dir):
        self.load_predefined_classes(default_prefdef_class_file)
        self.label_dialog = LabelDialog(parent=self, list_item=self.label_hist)
        self.color_dialog = ColorDialog(parent=self)
        self.zoom_widget = ZoomWidget(title=self.get_str('zoomin'))
        self.light_widget = LightWidget(self.get_str('lightWidgetTitle'))

        self.canvas = Canvas(parent=self)
        self.canvas.set_drawing_shape_to_square(self.settings.get(SETTING_DRAW_SQUARE, False))

        # 恢复设置等逻辑保持在 handlers 中或这里少量保留
        self.handlers.init_from_settings()

        if default_filename:
            if os.path.isdir(default_filename):
                QTimer.singleShot(0, partial(self.handlers.import_dir_images, default_filename))
            else:
                QTimer.singleShot(0, partial(self.handlers.load_file, default_filename))

    # ==================== 以下是需要暴露给外部或重写的少量方法 ====================

    def get_str(self, key: str, default=None, **kwargs):
        return self.i18n.tr(key, default=default, **kwargs)

    def tr(self, key: str, default=None, **kwargs):
        return self.get_str(key, default, **kwargs)

    def on_language_changed(self, lang_code: str):
        self.handlers.on_language_changed(lang_code)

    def set_dirty(self):
        self.handlers.set_dirty()

    def set_clean(self):
        self.handlers.set_clean()

    def may_continue(self):
        return self.handlers.may_continue()

    def closeEvent(self, event):
        if not self.may_continue():
            event.ignore()
            return
        self.handlers.save_settings()
        super().closeEvent(event)

    # 其他你原来在 MainWindow 里的快捷方法可以继续保留或转交给 handler
    def beginner(self):
        return self._beginner

    def advanced(self):
        return not self._beginner
